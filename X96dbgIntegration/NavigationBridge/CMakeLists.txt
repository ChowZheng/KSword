cmake_minimum_required(VERSION 3.20)
project(KSwordX64DbgNavigation LANGUAGES CXX)
add_library(KSwordNavigation SHARED NavigationBridge.cpp NavigationBridge.def)
target_compile_features(KSwordNavigation PRIVATE cxx_std_17)
target_compile_definitions(KSwordNavigation PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX UNICODE _UNICODE)
target_link_libraries(KSwordNavigation PRIVATE advapi32 user32)
if(MSVC)
  target_compile_options(KSwordNavigation PRIVATE /W4 /WX /utf-8)
  set_property(TARGET KSwordNavigation PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()

option(KSWORD_NAVIGATION_TESTS "Build the isolated fake-SDK navigation regression" OFF)
if(KSWORD_NAVIGATION_TESTS AND CMAKE_SIZEOF_VOID_P EQUAL 8)
  enable_testing()
  add_library(NavigationFakeSdk SHARED tests/FakeSdk.cpp)
  set_target_properties(NavigationFakeSdk PROPERTIES PREFIX "" OUTPUT_NAME "x64bridge")
  add_executable(NavigationTests tests/NavigationTests.cpp)
  foreach(target NavigationFakeSdk NavigationTests)
    target_compile_features(${target} PRIVATE cxx_std_17)
    target_compile_definitions(${target} PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX UNICODE _UNICODE)
    if(MSVC)
      target_compile_options(${target} PRIVATE /W4 /WX /utf-8)
      set_property(TARGET ${target} PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    endif()
  endforeach()
  add_dependencies(NavigationTests KSwordNavigation NavigationFakeSdk)
  add_custom_command(TARGET NavigationFakeSdk POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy
    $<TARGET_FILE:NavigationFakeSdk> $<TARGET_FILE_DIR:NavigationFakeSdk>/x64dbg.dll)
  add_test(NAME NavigationProtocol COMMAND NavigationTests)
  set_tests_properties(NavigationProtocol PROPERTIES WORKING_DIRECTORY $<TARGET_FILE_DIR:NavigationTests>)
endif()
if(CMAKE_SIZEOF_VOID_P EQUAL 8)
  set_target_properties(KSwordNavigation PROPERTIES PREFIX "" SUFFIX ".dp64")
else()
  set_target_properties(KSwordNavigation PROPERTIES PREFIX "" SUFFIX ".dp32")
endif()
